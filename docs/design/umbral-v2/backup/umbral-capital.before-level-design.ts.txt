// Coordinates are world units; an object's (x, y) is the centre of its base.

const house = (id: string, asset: 'house-a' | 'house-b', x: number, y: number): MapObject =>
  ({ id, asset, x, y, footprint: asset === 'house-a' ? { w: 4, h: 2.5 } : { w: 2.8, h: 2.5 } });
const prop = (id: string, asset: string, x: number, y: number, w = 0.8, h = 0.6): MapObject =>
  ({ id, asset, x, y, footprint: { w, h } });

const crates = (id: string, x: number, y: number) => prop(id, 'crates', x, y, 1.8, 1);
const barrels = (id: string, x: number, y: number) => prop(id, 'barrels', x, y, 1.6, 0.9);
const tree = (id: string, x: number, y: number) => prop(id, 'dead-tree', x, y, 1.2, 0.8);
const lantern = (id: string, x: number, y: number) => prop(id, 'skull-lantern', x, y);
